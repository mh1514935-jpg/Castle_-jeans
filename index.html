import sqlite3 import tkinter as tk from tkinter import messagebox, ttk import customtkinter as ctk # ضبط مظهر الواجهة ctk.set_appearance_mode("Dark") ctk.set_default_color_theme("blue") # --- إنشاء وقاعدة البيانات --- def init_db(): conn = sqlite3.connect("castle_jeans.db") cursor = conn.cursor() # جدول المنتجات والمخزون cursor.execute( """ CREATE TABLE IF NOT EXISTS inventory ( id INTEGER PRIMARY KEY AUTOINCREMENT, code TEXT UNIQUE, name TEXT, size TEXT, buy_price REAL, sell_price REAL, quantity INTEGER ) """ ) # جدول المبيعات cursor.execute( """ CREATE TABLE IF NOT EXISTS sales ( id INTEGER PRIMARY KEY AUTOINCREMENT, product_id INTEGER, quantity INTEGER, total_sell REAL, total_cost REAL, date TIMESTAMP DEFAULT CURRENT_TIMESTAMP ) """ ) # جدول المصاريف cursor.execute( """ CREATE TABLE IF NOT EXISTS expenses ( id INTEGER PRIMARY KEY AUTOINCREMENT, title TEXT, amount REAL, date TIMESTAMP DEFAULT CURRENT_TIMESTAMP ) """ ) conn.commit() conn.close() init_db() # --- الواجهة الرئيسية --- class CastleJeansApp(ctk.CTk): def __init__(self): super().__init__() self.title("Castle Jeans - نظام إدارة المبيعات والمخزون") self.geometry("1100x700") # العنوان الرئيسي self.header_label = ctk.CTkLabel( self, text="🏰 Castle Jeans", font=ctk.CTkFont(size=28, weight="bold"), ) self.header_label.pack(pady=15) # إنشاء التبويبات (Tabs) self.tabview = ctk.CTkTabview(self, width=1050, height=600) self.tabview.pack(padx=20, pady=10, fill="both", expand=True) self.tab_pos = self.tabview.add("🛒 المبيعات (POS)") self.tab_inventory = self.tabview.add("📦 مدخلات البضاعة والمخزن") self.tab_expenses = self.tabview.add("💸 المصاريف (إيجار/عمالة)") self.tab_reports = self.tabview.add("📊 التقارير وصافي الربح") # إعداد الأقسام self.setup_pos_tab() self.setup_inventory_tab() self.setup_expenses_tab() self.setup_reports_tab() # --- 1. تبويب المبيعات --- def setup_pos_tab(self): frame = self.tab_pos ctk.CTkLabel( frame, text="كود القطعة:", font=ctk.CTkFont(size=16) ).grid(row=0, column=0, padx=10, pady=10) self.code_sale_entry = ctk.CTkEntry(frame, width=200) self.code_sale_entry.grid(row=0, column=1, padx=10, pady=10) ctk.CTkLabel(frame, text="الكمية:", font=ctk.CTkFont(size=16)).grid( row=0, column=2, padx=10, pady=10 ) self.qty_sale_entry = ctk.CTkEntry(frame, width=100) self.qty_sale_entry.insert(0, "1") self.qty_sale_entry.grid(row=0, column=3, padx=10, pady=10) btn_sell = ctk.CTkButton( frame, text="إتمام عملية البيع 🛍️", fg_color="green", command=self.process_sale, ) btn_sell.grid(row=0, column=4, padx=15, pady=10) def process_sale(self): code = self.code_sale_entry.get() try: qty = int(self.qty_sale_entry.get()) except ValueError: messagebox.showerror("خطأ", "برجاء إدخال كمية صحيحة") return conn = sqlite3.connect("castle_jeans.db") cursor = conn.cursor() cursor.execute("SELECT id, buy_price, sell_price, quantity FROM inventory WHERE code=?", (code,)) item = cursor.fetchone() if not item: messagebox.showerror("خطأ", "المنتج غير موجود!") elif item[3] < qty: messagebox.showwarning("تنبيه", f"الكمية المتاحة في المخزن فقط {item[3]}") else: new_qty = item[3] - qty cursor.execute("UPDATE inventory SET quantity=? WHERE id=?", (new_qty, item[0])) total_sell = item[2] * qty total_cost = item[1] * qty cursor.execute("INSERT INTO sales (product_id, quantity, total_sell, total_cost) VALUES (?, ?, ?, ?)", (item[0], qty, total_sell, total_cost)) conn.commit() messagebox.showinfo("نجاح", f"تم البيع بنجاح! الإجمالي: {total_sell} ج.م") self.code_sale_entry.delete(0, 'end') self.refresh_inventory() conn.close() # --- 2. تبويب إضافة البضاعة والمخزن --- def setup_inventory_tab(self): frame = self.tab_inventory # نموذج الإدخال fields = [("الكود (الباركود)", 0), ("اسم القطعة", 1), ("المقاس", 2), ("سعر الشراء (التكلفة)", 3), ("سعر البيع", 4), ("الكمية", 5)] self.inv_entries = {} for label_text, row in fields: ctk.CTkLabel(frame, text=label_text).grid(row=row, column=0, padx=10, pady=5, sticky="e") entry = ctk.CTkEntry(frame, width=200) entry.grid(row=row, column=1, padx=10, pady=5) self.inv_entries[label_text] = entry btn_add = ctk.CTkButton(frame, text="إضافة / تحديث المخزون 📦", command=self.add_to_inventory) btn_add.grid(row=6, column=0, columnspan=2, pady=15) # جدول عرض المخزون self.tree = ttk.Treeview(frame, columns=("code", "name", "size", "buy", "sell", "qty"), show="headings", height=10) self.tree.heading("code", text="الكود") self.tree.heading("name", text="الاسم") self.tree.heading("size", text="المقاس") self.tree.heading("buy", text="سعر الشراء") self.tree.heading("sell", text="سعر البيع") self.tree.heading("qty", text="الكمية بالمخزن") self.tree.grid(row=0, column=2, rowspan=7, padx=20, pady=5) self.refresh_inventory() def add_to_inventory(self): try: code = self.inv_entries["الكود (الباركود)"].get() name = self.inv_entries["اسم القطعة"].get() size = self.inv_entries["المقاس"].get() buy = float(self.inv_entries["سعر الشراء (التكلفة)"].get()) sell = float(self.inv_entries["سعر البيع"].get()) qty = int(self.inv_entries["الكمية"].get()) conn = sqlite3.connect("castle_jeans.db") cursor = conn.cursor() cursor.execute(""" INSERT INTO inventory (code, name, size, buy_price, sell_price, quantity) VALUES (?, ?, ?, ?, ?, ?) ON CONFLICT(code) DO UPDATE SET quantity = quantity + excluded.quantity, buy_price = excluded.buy_price, sell_price = excluded.sell_price """, (code, name, size, buy, sell, qty)) conn.commit() conn.close() messagebox.showinfo("تم", "تمت إضافة البضاعة بنجاح!") self.refresh_inventory() except Exception as e: messagebox.showerror("خطأ", "تأكد من إدخال البيانات بشكل صحيح") def refresh_inventory(self): for item in self.tree.get_children(): self.tree.delete(item) conn = sqlite3.connect("castle_jeans.db") cursor = conn.cursor() cursor.execute("SELECT code, name, size, buy_price, sell_price, quantity FROM inventory") for row in cursor.fetchall(): self.tree.insert("", "end", values=row) conn.close() # --- 3. تبويب المصاريف (إيجار/عمالة) --- def setup_expenses_tab(self): frame = self.tab_expenses ctk.CTkLabel(frame, text="بيان المصروف (إيجار/كهرباء/عمالة):").grid(row=0, column=0, padx=10, pady=10) self.exp_title_entry = ctk.CTkEntry(frame, width=200) self.exp_title_entry.grid(row=0, column=1, padx=10, pady=10) ctk.CTkLabel(frame, text="المبلغ:").grid(row=1, column=0, padx=10, pady=10) self.exp_amount_entry = ctk.CTkEntry(frame, width=200) self.exp_amount_entry.grid(row=1, column=1, padx=10, pady=10) btn_add_exp = ctk.CTkButton(frame, text="تسجيل المصروف 💵", fg_color="orange", command=self.add_expense) btn_add_exp.grid(row=2, column=0, columnspan=2, pady=15) def add_expense(self): title = self.exp_title_entry.get() try: amount = float(self.exp_amount_entry.get()) conn = sqlite3.connect("castle_jeans.db") cursor = conn.cursor() cursor.execute("INSERT INTO expenses (title, amount) VALUES (?, ?)", (title, amount)) conn.commit() conn.close() messagebox.showinfo("تم", "تم تسجيل المصروف بنجاح!") self.exp_title_entry.delete(0, 'end') self.exp_amount_entry.delete(0, 'end') except ValueError: messagebox.showerror("خطأ", "يرجى كتابة مبلغ صحيح") # --- 4. تبويب الحسابات وصافي الربح --- def setup_reports_tab(self): frame = self.tab_reports btn_calc = ctk.CTkButton(frame, text="تحديث وتجميع الأرقام 🔄", command=self.calculate_financials) btn_calc.pack(pady=15) self.lbl_sales = ctk.CTkLabel(frame, text="إجمالي المبيعات: 0 ج.م", font=ctk.CTkFont(size=18)) self.lbl_sales.pack(pady=10) self.lbl_gross_profit = ctk.CTkLabel(frame, text="مجمل ربح البضاعة (البيع - التكلفة): 0 ج.م", font=ctk.CTkFont(size=18)) self.lbl_gross_profit.pack(pady=10) self.lbl_expenses = ctk.CTkLabel(frame, text="إجمالي المصاريف (إيجار/عمالة/غيرها): 0 ج.م", font=ctk.CTkFont(size=18)) self.lbl_expenses.pack(pady=10) self.lbl_net_profit = ctk.CTkLabel(frame, text="💰 صافي الربح النهائي: 0 ج.م", font=ctk.CTkFont(size=22, weight="bold"), text_color="green") self.lbl_net_profit.pack(pady=20) def calculate_financials(self): conn = sqlite3.connect("castle_jeans.db") cursor = conn.cursor() # إجمالي المبيعات والتكلفة cursor.execute("SELECT SUM(total_sell), SUM(total_cost) FROM sales") sales_data = cursor.fetchone() total_sales = sales_data[0] if sales_data[0] else 0.0 total_cost = sales_data[1] if sales_data[1] else 0.0 gross_profit = total_sales - total_cost # إجمالي المصاريف cursor.execute("SELECT SUM(amount) FROM expenses") exp_data = cursor.fetchone() total_expenses = exp_data[0] if exp_data[0] else 0.0 # صافي الربح net_profit = gross_profit - total_expenses # تحديث الواجهة self.lbl_sales.configure(text=f"إجمالي المبيعات: {total_sales:,.2f} ج.م") self.lbl_gross_profit.configure(text=f"مجمل ربح البضاعة: {gross_profit:,.2f} ج.م") self.lbl_expenses.configure(text=f"إجمالي المصاريف (إيجار/عمالة): {total_expenses:,.2f} ج.م") color = "green" if net_profit >= 0 else "red" self.lbl_net_profit.configure(text=f"💰 صافي الربح النهائي: {net_profit:,.2f} ج.م", text_color=color) conn.close() if __name__ == "__main__": app = CastleJeansApp() app.mainloop()
